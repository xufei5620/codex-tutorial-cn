        declared = manifest_paths_from_zip(self.zip_path)

def write_text(path, text):
    normalized = text.replace("\r\n", "\n").replace("\r", "\n").rstrip("\n") + "\n"
    with open(path, "w", encoding="utf-8", newline="\n") as handle:
        handle.write(normalized)
